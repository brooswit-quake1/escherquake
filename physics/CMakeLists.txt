cmake_minimum_required(VERSION 3.20)
project(EscherJolt LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(INTERPROCEDURAL_OPTIMIZATION OFF CACHE BOOL "" FORCE)
set(GENERATE_DEBUG_SYMBOLS OFF CACHE BOOL "" FORCE)
set(ENABLE_OBJECT_STREAM OFF CACHE BOOL "" FORCE)
set(ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
set(DEBUG_RENDERER_IN_DEBUG_AND_RELEASE OFF CACHE BOOL "" FORCE)
set(PROFILER_IN_DEBUG_AND_RELEASE OFF CACHE BOOL "" FORCE)
set(USE_SSE4_1 OFF CACHE BOOL "" FORCE)
set(USE_SSE4_2 OFF CACHE BOOL "" FORCE)
set(USE_AVX OFF CACHE BOOL "" FORCE)
set(USE_AVX2 OFF CACHE BOOL "" FORCE)
set(USE_LZCNT OFF CACHE BOOL "" FORCE)
set(USE_TZCNT OFF CACHE BOOL "" FORCE)
set(USE_F16C OFF CACHE BOOL "" FORCE)
set(USE_FMADD OFF CACHE BOOL "" FORCE)
add_subdirectory(../vendor/jolt/Build jolt)
set(FTE ${CMAKE_CURRENT_SOURCE_DIR}/../vendor/fteqw)
add_library(fteplug_jolt MODULE ${FTE}/plugins/plugin.c ${FTE}/plugins/jolt/jolt.cpp)
target_include_directories(fteplug_jolt PRIVATE ${FTE}/engine ${FTE}/engine/common ${FTE}/engine/client ${FTE}/engine/server ${FTE}/engine/gl ${FTE}/engine/qclib)
target_compile_definitions(fteplug_jolt PRIVATE FTEPLUGIN MULTITHREAD)
target_link_libraries(fteplug_jolt PRIVATE Jolt)
set_target_properties(fteplug_jolt PROPERTIES PREFIX "" C_VISIBILITY_PRESET hidden CXX_VISIBILITY_PRESET hidden)
if(UNIX AND NOT APPLE)
  target_link_options(fteplug_jolt PRIVATE -Wl,--no-undefined)
endif()

install(TARGETS fteplug_jolt LIBRARY DESTINATION . RUNTIME DESTINATION .)
